# Behavior Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A `behavior/` package and `tools/behave.py` that make the head find, follow, glance at and lean toward people using perception snapshots and the neck controller, with `calm` / `curious` / `shy` presets.

**Architecture:** A pure, clock-injected `Engine` state machine (attention → follow/idle → command) is driven by a 20 Hz `Runner` that reads `Perception.latest()` and calls `NeckController.look()`. Following is closed-loop: `pose at capture + gain × camera offset`, once per new snapshot. Everything is tested offline in a simulated room/camera/neck before the robot moves.

**Tech Stack:** Python 3.12, existing `neck/` and `perception/` packages, PyYAML, OpenCV (view), pytest.

**Spec:** `docs/superpowers/specs/2026-09-26-behavior-design.md`

## Global Constraints

- The engine and its helpers (`attention.py`, `follow.py`, `idle.py`, `engine.py`) do no I/O and take `now` as an argument.
- Behavior never clears neck faults automatically; only the `c` key in `--view` calls `clear_fault()`.
- Commands go through `NeckController.look()` only (it clamps to soft limits: nod ≥ 0, lean ≥ 0).
- Follow uses each snapshot **once** (keyed by `Snapshot.timestamp`); base pose = `Snapshot.neck_pose`, else the neck's pose, else the engine's last command, else `(0, 0, 0)`.
- Signs: +turn = left = +camera yaw; +nod = up = +camera pitch.
- Anything that opens the camera runs from **macOS Terminal.app** (the Claude app is denied camera access).
- Tests must not depend on the tunable `behavior/presets.yaml` values except the preset-loading test; use the `make_preset` fixture.

## Review Focus

1. **Camera fails or snapshots go stale while tracking** → treated as nobody there: `LOST`, then `IDLE`; never following a stale offset. Test: Task 6 `test_camera_failure_behaves_like_nobody_there`.
2. **Target beyond the neck's turn range** (e.g. 75° left) → neck sits at its limit without wind-up, and follows promptly when the person comes back into range. Test: Task 6 `test_target_beyond_turn_limit_does_not_wind_up`.
3. **Person below the camera** (negative pitch; nod can't go below level) → nod stays at 0, tracking continues, no fault. Test: Task 6 `test_person_below_level_keeps_nod_at_zero`.
4. **Person with no distance estimate** → closeness 0, never triggers lean, no crash. Tests: Task 4 `test_unknown_distance_scores_zero_closeness`, Task 5 `test_unknown_distance_never_leans`.
5. **An exception inside a runner tick** (e.g. perception or neck error) → logged, loop keeps running. Test: Task 7 `test_run_loop_survives_a_failing_tick`.

```python
"""Behavior presets: loading and validation."""
from __future__ import annotations

from dataclasses import dataclass, fields
from pathlib import Path

import yaml

PRESETS_FILE = Path(__file__).with_name("presets.yaml")


@dataclass(frozen=True)
class Preset:
    name: str
    w_facing: float
    w_near: float
    w_approach: float
    stickiness: float
    switch_hold_s: float
    follow_gain: float
    glance_newcomer: bool
    glance_movement: bool
    glance_s: float
    glance_cooldown_s: float
    lost_hold_s: float
    idle_scan: bool
    scan_after_s: float
    scan_amplitude_deg: float
    scan_speed_dps: float
    scan_nod_deg: float
    scan_pause_s: float
    lean_deg: float
    avert_after_s: float | None
    avert_deg: float
    avert_s: float
    deadband_deg: float
    lean_near_m: float
    lean_release_m: float
    lean_release_s: float
    coast_factor: float
    near_full_m: float
    near_zero_m: float
    newcomer_window_s: float
    loop_hz: float


def _validate(p: Preset) -> None:
    checks = [
        (0 < p.follow_gain <= 1, "follow_gain must be in (0, 1]"),
        (min(p.w_facing, p.w_near, p.w_approach, p.stickiness) >= 0,
         "weights and stickiness must be non-negative"),
        (p.near_full_m < p.near_zero_m, "near_full_m must be less than near_zero_m"),
        (p.lean_near_m < p.lean_release_m, "lean_near_m must be less than lean_release_m"),
        (p.lean_deg >= 0, "lean_deg must be non-negative"),
        (p.avert_after_s is None or p.avert_after_s > 0, "avert_after_s must be null or positive"),
        (0 <= p.coast_factor <= 1, "coast_factor must be within 0..1"),
        (all(v > 0 for v in (p.switch_hold_s, p.glance_s, p.glance_cooldown_s, p.lost_hold_s,
                             p.scan_after_s, p.scan_amplitude_deg, p.scan_speed_dps, p.scan_pause_s,
                             p.avert_s, p.deadband_deg, p.lean_release_s, p.newcomer_window_s,
                             p.loop_hz)),
         "times, speeds and amplitudes must be positive"),
    ]
    for ok, message in checks:
        if not ok:
            raise ValueError(f"preset {p.name}: {message}")


def preset_from_dict(name: str, d: dict) -> Preset:
    names = {f.name for f in fields(Preset)} - {"name"}
    missing = sorted(names - set(d))
    if missing:
        raise ValueError(f"preset {name} missing: {', '.join(missing)}")
    unknown = sorted(set(d) - names)
    if unknown:
        raise ValueError(f"preset {name} has unknown keys: {', '.join(unknown)}")
    preset = Preset(name=name, **{k: d[k] for k in names})
    _validate(preset)
    return preset


def load_presets(path: Path | str = PRESETS_FILE) -> dict[str, Preset]:
    raw = yaml.safe_load(Path(path).read_text())
    shared = raw.get("shared", {})
    return {name: preset_from_dict(name, {**shared, **values})
            for name, values in raw["presets"].items()}


def load_preset(name: str, path: Path | str = PRESETS_FILE) -> Preset:
    presets = load_presets(path)
    if name not in presets:
        raise ValueError(f"unknown preset {name!r}; choose from {', '.join(sorted(presets))}")
    return presets[name]
```

```python
"""Who to look at: scoring, stickiness with a sustained-lead switch rule, and glances."""
from __future__ import annotations

MIN_CONFIDENCE = 0.5


def closeness(distance_m, preset) -> float:
    if distance_m is None:
        return 0.0
    span = preset.near_zero_m - preset.near_full_m
    return min(1.0, max(0.0, (preset.near_zero_m - distance_m) / span))


def score(person, preset, is_target: bool = False) -> float:
    s = (preset.w_facing * person.facing_score
         + preset.w_near * closeness(person.distance_m, preset)
         + preset.w_approach * (1.0 if person.motion == "approaching" else 0.0))
    if not person.visible:
        s *= preset.coast_factor
    if is_target:
        s += preset.stickiness
    return s


class Attention:
    def __init__(self, preset):
        self.p = preset
        self.target_id: int | None = None
        self._challenger: int | None = None
        self._challenger_since = 0.0
        self._last_glance_at = float("-inf")

    def _eligible(self, people):
        return [x for x in people if x.confidence >= MIN_CONFIDENCE]

    def update(self, people, now: float) -> int | None:
        eligible = self._eligible(people)
        by_id = {x.id: x for x in eligible}
        if self.target_id not in by_id:
            self.target_id = None
            self._challenger = None
        if self.target_id is None:
            if eligible:
                self.target_id = max(eligible, key=lambda x: score(x, self.p)).id
            return self.target_id
        current = score(by_id[self.target_id], self.p, is_target=True)
        rivals = [x for x in eligible if x.id != self.target_id]
        best = max(rivals, key=lambda x: score(x, self.p), default=None)
        if best is not None and score(best, self.p) > current:
            if self._challenger != best.id:
                self._challenger, self._challenger_since = best.id, now
            elif now - self._challenger_since >= self.p.switch_hold_s:
                self.target_id, self._challenger = best.id, None
        else:
            self._challenger = None
        return self.target_id

    def glance_candidate(self, people, now: float) -> int | None:
        if now - self._last_glance_at < self.p.glance_cooldown_s:
            return None
        for x in self._eligible(people):
            if x.id == self.target_id:
                continue
            newcomer = self.p.glance_newcomer and now - x.first_seen <= self.p.newcomer_window_s
            moving = self.p.glance_movement and x.motion == "passing"
            if newcomer or moving:
                self._last_glance_at = now
                return x.id
        return None
```

"""Closed-loop follow law and lean-in policy."""
from __future__ import annotations

from dataclasses import dataclass


@dataclass(frozen=True)
class Command:
    turn: float
    nod: float
    lean: float


def follow_axis(base: float, offset: float, gain: float, deadband: float, previous) -> float:
    """Move part of the way from where the head was at capture toward the target."""
    if previous is not None and abs(offset) < deadband:
        return previous
    return base + gain * offset


class Lean:
    """Lean in when the target is close and facing; release with hysteresis."""

    def __init__(self, preset):
        self.p = preset
        self.active = False
        self._not_facing_since: float | None = None

    def reset(self) -> None:
        self.active = False
        self._not_facing_since = None

    def update(self, person, now: float) -> float:
        d = person.distance_m
        if not self.active:
            if d is not None and d <= self.p.lean_near_m and person.facing:
                self.active = True
                self._not_facing_since = None
        elif d is None or d > self.p.lean_release_m:
            self.reset()
        elif not person.facing:
            if self._not_facing_since is None:
                self._not_facing_since = now
            elif now - self._not_facing_since >= self.p.lean_release_s:
                self.reset()
        else:
            self._not_facing_since = None
        return self.p.lean_deg if self.active else 0.0

"""Idle scanning: sweep turn back and forth, pausing at each end."""
from __future__ import annotations


class Scanner:
    def __init__(self, preset):
        self.p = preset
        self.angle = 0.0
        self._direction = 1.0          # start sweeping left
        self._pause_until = float("-inf")

    def reset(self, start_deg: float = 0.0) -> None:
        self.angle = max(-self.p.scan_amplitude_deg, min(self.p.scan_amplitude_deg, start_deg))
        self._direction = 1.0
        self._pause_until = float("-inf")

    def step(self, dt: float, now: float) -> float:
        if now < self._pause_until:
            return self.angle
        self.angle += self._direction * self.p.scan_speed_dps * dt
        if abs(self.angle) >= self.p.scan_amplitude_deg:
            self.angle = self._direction * self.p.scan_amplitude_deg
            self._direction = -self._direction
            self._pause_until = now + self.p.scan_pause_s
        return self.angle

"""Closed-loop simulation: people in the room, a camera on the moving head, a speed-limited neck."""
from __future__ import annotations

from dataclasses import dataclass

from behavior.engine import Engine, NeckStatus
from perception.pipeline import people_or_empty
from perception.types import Person, Snapshot

HFOV, VFOV = 60.0, 34.0


@dataclass
class SimPerson:
    id: int
    yaw: float                   # room direction in degrees, +left
    pitch: float = 10.0
    distance: float = 2.0
    facing_score: float = 0.9
    motion: str = "still"
    first_seen: float = -10.0
    yaw_rate: float = 0.0


class SimNeck:
    LIMITS = {"turn": (-60.0, 60.0), "nod": (0.0, 60.0), "lean": (0.0, 30.0)}
    SPEED = {"turn": 120.0, "nod": 30.0, "lean": 20.0}

    def __init__(self):
        self.pos = {"turn": 0.0, "nod": 0.0, "lean": 0.0}
        self.target = dict(self.pos)
        self.running = True
        self.looks = []

    def look(self, turn, nod, lean) -> bool:
        if not self.running:
            return False
        for k, v in (("turn", turn), ("nod", nod), ("lean", lean)):
            lo, hi = self.LIMITS[k]
            self.target[k] = min(max(v, lo), hi)
        self.looks.append((turn, nod, lean))
        return True

    def step(self, dt) -> None:
        for k in self.pos:
            m = self.SPEED[k] * dt
            self.pos[k] += max(-m, min(m, self.target[k] - self.pos[k]))

    def pose(self):
        return (self.pos["turn"], self.pos["nod"], self.pos["lean"])


class SimCamera:
    def __init__(self, fps=19.0, latency=0.06):
        self.period, self.latency = 1.0 / fps, latency
        self._next = 0.0
        self._pending = []
        self.failed = False
        self.latest = Snapshot(0.0, (1280, 720), (), False, "not started")

    def step(self, now, world, neck) -> None:
        if now >= self._next:
            pose = neck.pose()
            if self.failed:
                shot = Snapshot(now, (1280, 720), (), False, "no frames from source")
            else:
                seen = tuple(p for p in (self._see(w, pose, now) for w in world) if p is not None)
                shot = Snapshot(now, (1280, 720), seen, True, None, pose)
            self._pending.append((now + self.latency, shot))
            self._next += self.period
        while self._pending and self._pending[0][0] <= now:
            self.latest = self._pending.pop(0)[1]

    @staticmethod
    def _see(w, pose, now):
        yaw, pitch = w.yaw - pose[0], w.pitch - pose[1]
        if abs(yaw) > HFOV / 2 or abs(pitch) > VFOV / 2:
            return None
        return Person(id=w.id, box=(0.0, 0.0, 1.0, 1.0), head_px=(0.0, 0.0), head_deg=(yaw, pitch),
                      world_yaw_deg=w.yaw, distance_m=w.distance, distance_source="body",
                      facing_score=w.facing_score, facing=w.facing_score >= 0.6, motion=w.motion,
                      confidence=0.9, visible=True, first_seen=w.first_seen, last_seen=now)


class Sim:
    def __init__(self, preset, people=(), dt=0.05):
        self.engine = Engine(preset)
        self.neck = SimNeck()
        self.camera = SimCamera()
        self.people = list(people)
        self.dt = dt
        self.t = 0.0
        self.log = []

    def run(self, seconds: float) -> None:
        for _ in range(round(seconds / self.dt)):
            self.t = round(self.t + self.dt, 9)
            for p in self.people:
                p.yaw += p.yaw_rate * self.dt
            self.camera.step(self.t, self.people, self.neck)
            shot = self.camera.latest
            people = people_or_empty(shot, self.t, 0.3)
            cmd = self.engine.tick(people, shot if shot.source_ok else None,
                                   NeckStatus(self.neck.running, self.neck.pose()), self.t)
            if cmd is not None:
                self.neck.look(cmd.turn, cmd.nod, cmd.lean)
            self.neck.step(self.dt)
            self.log.append((self.t, self.engine.state, self.neck.pose(), self.engine.attention.target_id))

    def turn(self) -> float:
        return self.neck.pos["turn"]

```python
"""Behavior state machine. Pure: no I/O, time is passed in."""
from __future__ import annotations

from dataclasses import dataclass
from enum import Enum

from behavior.attention import Attention
from behavior.follow import Command, Lean, follow_axis
from behavior.idle import Scanner


class State(str, Enum):
    TRACKING = "tracking"
    GLANCING = "glancing"
    AVERTING = "averting"
    LOST = "lost"
    IDLE = "idle"
    SCANNING = "scanning"
    PAUSED = "paused"


@dataclass(frozen=True)
class NeckStatus:
    running: bool                                  # neck RUNNING with no latched fault
    pose: tuple[float, float, float] | None


NEUTRAL = Command(0.0, 0.0, 0.0)


class Engine:
    def __init__(self, preset):
        self.p = preset
        self.attention = Attention(preset)
        self.lean = Lean(preset)
        self.scanner = Scanner(preset)
        self.state = State.IDLE
        self.command: Command | None = None
        self.glance_id: int | None = None
        self._since = 0.0
        self._last_tick: float | None = None
        self._last_snapshot_ts: float | None = None
        self._facing_since: float | None = None
        self._avert_command: Command | None = None

    def _enter(self, state: State, now: float) -> None:
        self.state = state
        self._since = now

    def _base(self, snapshot, neck):
        if snapshot is not None and snapshot.neck_pose is not None:
            return snapshot.neck_pose
        if neck.pose is not None:
            return neck.pose
        if self.command is not None:
            return (self.command.turn, self.command.nod, self.command.lean)
        return (0.0, 0.0, 0.0)

    def _follow(self, person, base, new_frame: bool, lean: float) -> Command:
        prev = self.command
        if not new_frame and prev is not None:
            return Command(prev.turn, prev.nod, lean)
        yaw, pitch = person.head_deg
        turn = follow_axis(base[0], yaw, self.p.follow_gain, self.p.deadband_deg,
                           prev.turn if prev else None)
        nod = follow_axis(base[1], pitch, self.p.follow_gain, self.p.deadband_deg,
                          prev.nod if prev else None)
        return Command(turn, nod, lean)

    def tick(self, people, snapshot, neck: NeckStatus, now: float) -> Command | None:
        dt = 0.0 if self._last_tick is None else now - self._last_tick
        self._last_tick = now

        if not neck.running:
            if self.state != State.PAUSED:
                self._enter(State.PAUSED, now)
            return None
        if self.state == State.PAUSED:          # resume from scratch
            self._enter(State.IDLE, now)
            self.command = None
            self.lean.reset()

        new_frame = snapshot is not None and snapshot.timestamp != self._last_snapshot_ts
        if new_frame:
            self._last_snapshot_ts = snapshot.timestamp
        base = self._base(snapshot, neck)
        by_id = {x.id: x for x in people}
        target_id = self.attention.update(people, now)

        # --- transitions ---
        if self.state in (State.TRACKING, State.GLANCING, State.AVERTING) and target_id is None:
            self._enter(State.LOST, now)
            self.lean.reset()
        if self.state in (State.LOST, State.IDLE, State.SCANNING) and target_id is not None:
            self._enter(State.TRACKING, now)
            self._facing_since = None
        if self.state == State.TRACKING:
            target = by_id[target_id]
            if self.p.avert_after_s is not None:
                if target.facing:
                    self._facing_since = now if self._facing_since is None else self._facing_since
                    if now - self._facing_since >= self.p.avert_after_s:
                        cur = self.command.turn if self.command else base[0]
                        away = self.p.avert_deg if cur <= 0 else -self.p.avert_deg
                        self._avert_command = Command(cur + away, self.command.nod if self.command else base[1], 0.0)
                        self._enter(State.AVERTING, now)
                else:
                    self._facing_since = None
            if self.state == State.TRACKING:
                glance = self.attention.glance_candidate(people, now)
                if glance is not None:
                    self.glance_id = glance
                    self._enter(State.GLANCING, now)
        if self.state == State.GLANCING and (self.glance_id not in by_id
                                             or now - self._since >= self.p.glance_s):
            self._enter(State.TRACKING, now)
        if self.state == State.AVERTING and now - self._since >= self.p.avert_s:
            self._enter(State.TRACKING, now)
            self._facing_since = None
        if self.state == State.LOST and now - self._since >= self.p.lost_hold_s:
            self._enter(State.IDLE, now)
        if (self.state == State.IDLE and self.p.idle_scan
                and now - self._since >= self.p.scan_after_s):
            self.scanner.reset(self.command.turn if self.command else 0.0)
            self._enter(State.SCANNING, now)

        # --- output ---
        if self.state == State.TRACKING:
            target = by_id[target_id]
            cmd = self._follow(target, base, new_frame, self.lean.update(target, now))
        elif self.state == State.GLANCING:
            lean = self.command.lean if self.command else 0.0
            cmd = self._follow(by_id[self.glance_id], base, new_frame, lean)
        elif self.state == State.AVERTING:
            cmd = self._avert_command
        elif self.state == State.LOST:
            cmd = self.command or NEUTRAL
        elif self.state == State.SCANNING:
            cmd = Command(self.scanner.step(dt, now), self.p.scan_nod_deg, 0.0)
        else:                                            # IDLE
            cmd = NEUTRAL

        changed = cmd != self.command
        self.command = cmd
        return cmd if changed else None
```

```python
"""20 Hz loop: perception snapshot -> engine -> neck (or printed commands in dry-run)."""
from __future__ import annotations

import logging
import threading
import time

from behavior.engine import NeckStatus
from perception.pipeline import people_or_empty

log = logging.getLogger("behavior.runner")


class Runner:
    def __init__(self, engine, perception, neck=None, stale_after_s: float = 0.3,
                 clock=time.monotonic, sleep=time.sleep, printer=print):
        self.engine = engine
        self.perception = perception
        self.neck = neck
        self.stale_after_s = stale_after_s
        self.clock = clock
        self._sleep = sleep
        self.printer = printer
        self._last_print = float("-inf")

    def neck_status(self) -> NeckStatus:
        if self.neck is None:                       # dry-run
            return NeckStatus(running=True, pose=None)
        from neck.controller import Mode

        st = self.neck.state
        running = st["mode"] == Mode.RUNNING and st["latched"] is None
        return NeckStatus(running=running, pose=self.neck.pose())

    def status_line(self) -> str:
        c = self.engine.command
        cmd = f"turn {c.turn:+6.1f} nod {c.nod:+5.1f} lean {c.lean:4.1f}" if c else "no command"
        return f"{self.engine.state.value:9s} target {self.engine.attention.target_id}  {cmd}"

    def tick(self):
        now = self.clock()
        snapshot = self.perception.latest()
        people = people_or_empty(snapshot, now, self.stale_after_s)
        cmd = self.engine.tick(people, snapshot if snapshot.source_ok else None,
                               self.neck_status(), now)
        if cmd is not None:
            if self.neck is not None:
                if not self.neck.look(cmd.turn, cmd.nod, cmd.lean):
                    log.warning("neck refused look(); pausing until it runs again")
            elif now - self._last_print >= 0.5:
                self.printer(self.status_line())
                self._last_print = now
        return cmd

    def run(self, stop: threading.Event) -> None:
        period = 1.0 / self.engine.p.loop_hz
        while not stop.is_set():
            start = self.clock()
            try:
                self.tick()
            except Exception:
                log.exception("behavior tick failed")
            self._sleep(max(0.0, period - (self.clock() - start)))

    def clear_fault(self) -> None:
        if self.neck is not None:
            self.neck.clear_fault()
```

```python
"""Run the head: perception -> behavior -> neck.

Usage (from macOS Terminal.app - the only place the camera is accessible):
  .venv/bin/python tools/behave.py --preset calm --dry-run    # decide, print, don't move
  .venv/bin/python tools/behave.py --preset calm --view       # move, with a debug window
Keys in --view: q quit, c clear a neck fault. Ctrl-C quits otherwise; the neck rests and holds.
"""
import argparse
import logging
import sys
import threading
from pathlib import Path

REPO = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(REPO))  # repo root, for package imports when run as a script


def main():
    parser = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    parser.add_argument("--preset", default="calm", help="calm, curious or shy")
    parser.add_argument("--view", action="store_true", help="show the camera with overlays")
    parser.add_argument("--dry-run", action="store_true", help="print decisions, don't move the neck")
    args = parser.parse_args()

    import cv2

    from behavior.config import load_preset
    from behavior.engine import Engine
    from behavior.runner import Runner
    from perception.config import load_config as load_perception_config
    from perception.detector import YoloPoseDetector
    from perception.pipeline import Perception
    from perception.sources import CameraSource
    from tools.perception_view import draw_overlay

    logging.basicConfig(level=logging.INFO)
    preset = load_preset(args.preset)
    pcfg = load_perception_config()
    neck = None
    if not args.dry_run:
        if input("This MOVES the neck. Hand on servo power? [y/N] ").strip().lower() != "y":
            return
        from neck.bus import ServoBus
        from neck.config import load_config as load_neck_config
        from neck.controller import NeckController

        ncfg = load_neck_config()
        neck = NeckController(ncfg, ServoBus.from_config(ncfg))
        neck.start()

    latest = {"snap": None, "img": None}
    perception = Perception(CameraSource.from_config(pcfg), YoloPoseDetector.from_config(pcfg, REPO),
                            pcfg, pose_provider=neck.pose if neck else None,
                            on_snapshot=lambda s, i: latest.update(snap=s, img=i))
    runner = Runner(Engine(preset), perception, neck, stale_after_s=pcfg.stale_after_s)
    stop = threading.Event()
    try:
        with perception:
            if args.view:
                loop = threading.Thread(target=runner.run, args=(stop,), daemon=True)
                loop.start()
                while True:
                    snap, img = latest["snap"], latest["img"]
                    if snap is not None and img is not None:
                        frame = draw_overlay(img.copy(), snap, 0.0, 0.0)
                        cv2.putText(frame, f"[{preset.name}] {runner.status_line()}", (10, 74),
                                    cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 255), 2)
                        cv2.imshow("behave", frame)
                    key = cv2.waitKey(20) & 0xFF
                    if key in (ord("q"), 27):
                        break
                    if key == ord("c"):
                        runner.clear_fault()
                stop.set()
                loop.join(timeout=1.0)
            else:
                print(f"[{preset.name}] running - Ctrl-C to stop")
                runner.run(stop)
    except KeyboardInterrupt:
        pass
    finally:
        stop.set()
        if neck is not None:
            neck.close()
        cv2.destroyAllWindows()


if __name__ == "__main__":
    main()
```
